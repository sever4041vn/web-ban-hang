CREATE DATABASE IF NOT EXISTS inventory_manager;
USE inventory_manager;

-- 1. Bảng danh mục sản phẩm
CREATE TABLE categories (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    description TEXT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 1.2. Đơn vị tính sản phẩm
CREATE TABLE unit (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    description TEXT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 2. Bảng sản phẩm
CREATE TABLE products (
    id INT AUTO_INCREMENT PRIMARY KEY,
    sku VARCHAR(50) UNIQUE NOT NULL, -- Mã sản phẩm duy nhất
    name VARCHAR(255) NOT NULL,
    category_id INT,
    cost_price DECIMAL(15, 2) DEFAULT 0, -- Giá vốn (để tính lợi nhuận)
    selling_price DECIMAL(15, 2) DEFAULT 0, -- Giá bán
    stock_quantity INT DEFAULT 0, -- Số lượng tồn kho hiện tại
    unit VARCHAR(20) DEFAULT 'Cái', -- Đơn vị tính
    FOREIGN KEY (category_id) REFERENCES categories(id) ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 3. Bảng nhà cung cấp (Dành cho việc nhập hàng)
CREATE TABLE suppliers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    phone VARCHAR(20),
    address TEXT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 4. Bảng khách hàng (Dành cho hóa đơn bán hàng)
CREATE TABLE customers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    phone VARCHAR(20),
    address TEXT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 5. Bảng Hóa đơn (Orders)
CREATE TABLE orders (
    id INT AUTO_INCREMENT PRIMARY KEY,
    invoice_no VARCHAR(50) UNIQUE NOT NULL, -- Số hóa đơn (VD: HD20240212001)
    customer_id INT,
    address VARCHAR(100) DEFAULT NULL,
    total_amount DECIMAL(15, 2) NOT NULL,
    discount DECIMAL(15, 2) DEFAULT 0,
    final_amount DECIMAL(15, 2) NOT NULL, -- Tổng tiền sau chiết khấu
    profit_amount DECIMAL(15, 2) NOT NULL, -- Tổng tiền sau chiết khấu
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (customer_id) REFERENCES customers(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 6. Chi tiết hóa đơn (Để lưu từng món trong 1 hóa đơn)
CREATE TABLE order_items (
    id INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT NOT NULL,
    product_id INT NOT NULL,
    unit VARCHAR(20) DEFAULT 'Cái',
    quantity INT NOT NULL,
    cost_price DECIMAL(15, 2) NOT NULL, -- Lưu giá tại thời điểm bán
    selling_price DECIMAL(15, 2) NOT NULL, -- Lưu giá tại thời điểm bán
    subtotal DECIMAL(15, 2) NOT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(id) ON DELETE CASCADE,
    FOREIGN KEY (product_id) REFERENCES products(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 7. Nhật ký kho (Lưu lịch sử mọi biến động Nhập/Xuất)
CREATE TABLE inventory_log (
    id INT AUTO_INCREMENT PRIMARY KEY,
    product_id INT NOT NULL,
    type ENUM('import', 'export', 'adjustment') NOT NULL, -- Nhập, Xuất, Điều chỉnh
    quantity INT NOT NULL,
    reference_id INT, -- ID của hóa đơn hoặc phiếu nhập tương ứng
    note TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (product_id) REFERENCES products(id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
